src/modules/triage/severity-engine.ts

import { hasRedFlags } from "./red-flags";

export type Severity = "LOW" | "MEDIUM" | "HIGH" | "EMERGENCY";

interface SeverityInput {
  symptoms: string[];
  duration?: string;
}

export function calculateSeverity({ symptoms, duration }: SeverityInput): Severity {
  // 1. Immediate Emergency Check
  if (hasRedFlags(symptoms)) {
    return "EMERGENCY";
  }

  // 2. Duration Parsing (mock logic, could use NLP in production)
  const durationStr = (duration || "").toLowerCase();
  const isLongDuration = durationStr.includes("week") || durationStr.includes("month");
  const isShortDuration = durationStr.includes("hour") || durationStr.includes("day");

  // 3. High Severity Check (Severe pain or long duration)
  const hasSeverePain = symptoms.some(s => s.toLowerCase().includes("severe"));
  if (hasSeverePain || isLongDuration) {
    return "HIGH";
  }

  // 4. Medium Severity Check (Fever, persistent pain)
  const hasFever = symptoms.some(s => s.toLowerCase().includes("fever"));
  const hasPain = symptoms.some(s => s.toLowerCase().includes("pain"));
  if (hasFever || (hasPain && isShortDuration)) {
    return "MEDIUM";
  }

  // 5. Default to Low
  return "LOW";
}

export type Route = "SELF_CARE" | "PHARMACY" | "CLINIC" | "HOSPITAL" | "EMERGENCY";

export function routeBySeverity(severity: Severity): Route {
  switch (severity) {
    case "EMERGENCY":
      return "EMERGENCY";
    case "HIGH":
      return "HOSPITAL";
    case "MEDIUM":
      return "CLINIC";
    case "LOW":
      return "PHARMACY";
    default:
      return "SELF_CARE";
  }
}

// Country specific overrides (e.g., in some countries, go to pharmacy first for fever)
export function applyCountryOverrides(route: Route, country: string, symptoms: string[]): Route {
  if (country === "Japan" && symptoms.includes("fever") && route === "CLINIC") {
    return "PHARMACY"; // In Japan, pharmacies often handle fever first
  }
  return route;
}

2. The Navigation Orchestrator (The Core Moat)
This module ties everything together. It takes raw symptoms, calculates the route, fetches medications, fetches providers, and builds the interpreter context.

src/modules/triage/navigation-orchestrator.ts

"use server"

import { prisma } from "@/lib/prisma";
import { calculateSeverity } from "./severity-engine";
import { routeBySeverity, applyCountryOverrides } from "./router";

export async function orchestrateNavigation(sessionId: string) {
  // 1. Fetch the TravelHealthSession
  const session = await prisma.travelHealthSession.findUnique({
    where: { id: sessionId },
    include: { user: { include: { memory: true } } }
  });

  if (!session) throw new Error("Session not found");

  // 2. Calculate Severity & Route
  const severity = calculateSeverity({
    symptoms: session.symptoms as string[],
    duration: session.duration || undefined
  });
  
  let route = routeBySeverity(severity);
  route = applyCountryOverrides(route, session.country, session.symptoms as string[]);

  // 3. Execute Route-Specific Actions
  let medicationRecommendations: any[] = [];
  let providerRecommendations: any[] = [];

  if (route === "PHARMACY" || route === "CLINIC") {
    // Mock: Fetch medication equivalents based on symptoms
    // In production, map symptoms -> active ingredients -> country equivalents
    medicationRecommendations = await prisma.countryMedication.findMany({
      where: { country: session.country },
      include: { activeIngredient: true },
      take: 3
    });

    // Fetch Providers
    providerRecommendations = await prisma.provider.findMany({
      where: { 
        country: session.country, 
        city: session.city,
        type: route === "PHARMACY" ? "PHARMACY" : "CLINIC"
      },
      take: 5
    });
  }

  if (route === "HOSPITAL" || route === "EMERGENCY") {
    providerRecommendations = await prisma.provider.findMany({
      where: { 
        country: session.country, 
        city: session.city,
        type: "HOSPITAL"
      },
      take: 5
    });
  }

  // 4. Update Session with Navigation Context
  const updatedSession = await prisma.travelHealthSession.update({
    where: { id: sessionId },
    data: {
      severityLevel: severity,
      providerOptions: providerRecommendations,
      localMedicationRecommendations: medicationRecommendations,
      interpreterContext: {
        severity,
        route,
        patientHistory: session.user.memory
      }
    }
  });

  return {
    severity,
    route,
    medications: medicationRecommendations,
    providers: providerRecommendations,
    session: updatedSession
  };
}

// Define the signature for an Agent Tool
type AgentTool = (args: any, context: { userId: string, sessionId?: string }) => Promise<any>;

// The Registry
export const toolRegistry: Record<string, AgentTool> = {
  
  queryPassport: async (_, ctx) => {
    return prisma.healthPassport.findFirst({ 
      where: { userId: ctx.userId },
      orderBy: { createdAt: "desc" }
    });
  },

  queryMemory: async (_, ctx) => {
    return prisma.medicalMemory.findMany({ where: { userId: ctx.userId } });
  },

  searchMedication: async (args, _) => {
    return prisma.countryMedication.findMany({
      where: {
        activeIngredient: { name: { equals: args.ingredient, mode: "insensitive" } },
        country: { equals: args.country, mode: "insensitive" }
      },
      include: { activeIngredient: true }
    });
  },

  findProvider: async (args, _) => {
    return prisma.provider.findMany({
      where: {
        country: { equals: args.country, mode: "insensitive" },
        city: { equals: args.city, mode: "insensitive" },
        type: args.type
      }
    });
  },

  // The ultimate tool: The Navigation Router
  navigateSymptoms: async (args, ctx) => {
    if (!ctx.sessionId) throw new Error("Session required for navigation");
    return orchestrateNavigation(ctx.sessionId);
  }
};

export async function executeTool(toolName: string, args: any, ctx: { userId: string, sessionId?: string }) {
  const tool = toolRegistry[toolName];
  if (!tool) throw new Error(`Tool ${toolName} not found in registry`);
  return tool(args, ctx);
}

"use server"

import { prisma } from "@/lib/prisma"
import { auth } from "@/lib/auth"

export async function getGlobalInsights(symptoms: string[], country: string) {
  const session = await auth()
  if (!session?.user?.id) return null

  // Find historical outcomes matching these symptoms in this country
  const outcomes = await prisma.outcome.findMany({
    where: {
      country: { equals: country, mode: "insensitive" },
      symptoms: { hasSome: symptoms }
    }
  })

  if (outcomes.length === 0) {
    return {
      totalCases: 0,
      recoveryRate: "0.0",
      avgRecoveryHours: 0,
      medicationData: [],
      recoveryData: [],
      providerData: []
    }
  }

  // 1. Basic Metrics
  const totalCases = outcomes.length
  const recoveredCases = outcomes.filter(o => o.result === "RECOVERED")
  const recoveryRate = ((recoveredCases.length / totalCases) * 100).toFixed(1)

  // 2. Recovery Time
  const validRecoveryTimes = recoveredCases.filter(o => o.recoveryTimeHours)
  const avgRecoveryHours = validRecoveryTimes.length > 0 
    ? Math.round(validRecoveryTimes.reduce((sum, o) => sum + (o.recoveryTimeHours || 0), 0) / validRecoveryTimes.length)
    : 0

  // 3. Medication Efficacy
  const medCounts: Record<string, { success: number; total: number }> = {}
  outcomes.forEach(o => {
    if (o.medicationUsed) {
      if (!medCounts[o.medicationUsed]) medCounts[o.medicationUsed] = { success: 0, total: 0 }
      medCounts[o.medicationUsed].total++
      if (o.result === "RECOVERED") medCounts[o.medicationUsed].success++
    }
  })
  
  const medicationData = Object.keys(medCounts).map(name => ({
    name,
    usage: medCounts[name].total,
    successRate: medCounts[name].total > 0 
      ? `${((medCounts[name].success / medCounts[name].total) * 100).toFixed(0)}%` 
      : "N/A"
  }))

  // 4. Recovery Distribution (for Pie Chart)
  const recoveryData = [
    { name: "Recovered", value: recoveredCases.length },
    { name: "No Change", value: outcomes.filter(o => o.result === "NO_CHANGE").length },
    { name: "Worsened", value: outcomes.filter(o => o.result === "WORSENED").length },
  ]

  // 5. Provider Success Rate
  const providerCounts: Record<string, { success: number; total: number }> = {}
  outcomes.forEach(o => {
    if (o.providerType) {
      if (!providerCounts[o.providerType]) providerCounts[o.providerType] = { success: 0, total: 0 }
      providerCounts[o.providerType].total++
      if (o.result === "RECOVERED") providerCounts[o.providerType].success++
    }
  })

  const providerData = Object.keys(providerCounts).map(type => ({
    name: type,
    successRate: providerCounts[type].total > 0 
      ? ((providerCounts[type].success / providerCounts[type].total) * 100).toFixed(1) 
      : "0.0"
  }))

  return {
    totalCases,
    recoveryRate,
    avgRecoveryHours,
    medicationData,
    recoveryData,
    providerData
  }
}

"use server"

import { auth } from "@/lib/auth"
import { prisma } from "@/lib/prisma"
import { revalidatePath } from "next/cache"

export async function captureOutcome(data: {
  sessionId: string;
  medicationUsed?: string;
  providerId?: string;
  providerType?: string;
  result: "RECOVERED" | "NO_CHANGE" | "WORSENED";
  recoveryTimeHours?: number;
}) {
  const session = await auth()
  if (!session?.user?.id) throw new Error("Unauthorized")

  const ths = await prisma.travelHealthSession.findUnique({ where: { id: data.sessionId } })
  if (!ths || ths.userId !== session.user.id) throw new Error("Invalid session")

  // 1. Create Outcome record
  await prisma.outcome.create({
    data: {
      travelHealthSessionId: data.sessionId,
      userId: session.user.id,
      symptoms: ths.symptoms as string[],
      medicationUsed: data.medicationUsed,
      providerId: data.providerId,
      providerType: data.providerType,
      country: ths.country,
      result: data.result,
      recoveryTimeHours: data.recoveryTimeHours,
    }
  })

  // 2. Update Session Status
  await prisma.travelHealthSession.update({
    where: { id: data.sessionId },
    data: { outcomeStatus: "COMPLETED" }
  })

  revalidatePath("/dashboard/outcomes")
  revalidatePath("/dashboard/journeys")
}

Instead of uploading files through Next.js Server Actions (which limits file size and blocks the event loop), the client requests a presigned URL from the action, then uploads directly to AWS S3. The action then records the metadata and audit log.

import { S3Client } from "@aws-sdk/client-s3";
import { getSignedUrl } from "@aws-sdk/s3-request-presigner";
import { PutObjectCommand } from "@aws-sdk/client-s3";

const s3 = new S3Client({
  region: process.env.AWS_REGION!,
  credentials: {
    accessKeyId: process.env.AWS_ACCESS_KEY_ID!,
    secretAccessKey: process.env.AWS_SECRET_ACCESS_KEY!,
  },
});

export async function generatePresignedUploadUrl(fileName: string, fileType: string) {
  const key = `vault/${Date.now()}-${fileName}`;
  const command = new PutObjectCommand({
    Bucket: process.env.AWS_S3_BUCKET_NAME!,
    Key: key,
    ContentType: fileType,
  });

  const url = await getSignedUrl(s3, command, { expiresIn: 60 }); // 1 minute expiry
  return { url, key };
}

import PDFDocument from "pdfkit";
import QRCode from "qrcode";
import { prisma } from "@/lib/prisma";

export async function generatePassportPDF(passportId: string): Promise<Buffer> {
  const passport = await prisma.healthPassport.findUnique({
    where: { id: passportId },
    include: { user: true }
  });

  if (!passport) throw new Error("Passport not found");

  const doc = new PDFDocument({ margin: 50 });
  const buffers: Buffer[] = [];
  doc.on("data", buffers.push.bind(buffers));

  return new Promise<Buffer>(async (resolve, reject) => {
    doc.on("end", () => {
      const pdfData = Buffer.concat(buffers);
      resolve(pdfData);
    });

    doc.on("error", reject);

    // Header
    doc.fontSize(20).fillColor("#2563eb").text("SwasthYatra Health Passport", { align: "center" });
    doc.moveDown();
    doc.fontSize(12).fillColor("#000").text(`Patient: ${passport.user.name}`, { align: "center" });
    doc.moveDown(2);

    // QR Code Generation
    const shareUrl = `${process.env.NEXT_PUBLIC_APP_URL}/api/passport/public/${passport.shareToken}`;
    const qrBuffer = await QRCode.toBuffer(shareUrl, { width: 150, margin: 1 });
    
    doc.image(qrBuffer, { fit: [150, 150], align: "center" });
    doc.moveDown(2);

    const data = passport.data as any;

    // Conditions
    doc.fontSize(14).fillColor("#dc2626").text("Conditions", { underline: true });
    doc.fontSize(11).fillColor("#000");
    data.conditions.forEach((c: any) => doc.text(`- ${c.name}`));
    doc.moveDown();

    // Allergies
    doc.fontSize(14).fillColor("#dc2626").text("Allergies", { underline: true });
    doc.fontSize(11).fillColor("#000");
    if (data.allergies.length === 0) doc.text("None reported");
    data.allergies.forEach((a: any) => doc.text(`- ${a.name}`));
    doc.moveDown();

    // Medications
    doc.fontSize(14).fillColor("#2563eb").text("Medications", { underline: true });
    doc.fontSize(11).fillColor("#000");
    if (data.medications.length === 0) doc.text("None reported");
    data.medications.forEach((m: any) => doc.text(`- ${m.name} ${m.dosage || ""}`));
    doc.moveDown();

    // Footer
    doc.fontSize(9).fillColor("#666").text(`Generated: ${new Date(passport.createdAt).toLocaleString()}`, { align: "center" });
    doc.text(`Expires: ${passport.expiresAt ? new Date(passport.expiresAt).toLocaleString() : 'N/A'}`, { align: "center" });

    doc.end();
  });
}

    }))
  })

  await Promise.all(memoryOps)

export async function inviteMember(email: string, role: string, orgId: string) {
  const session = await auth()
  if (!session?.user?.id) throw new Error("Unauthorized")

  // Verify inviter is Org Admin
  const membership = await prisma.organizationMember.findUnique({
    where: { userId_organizationId: { userId: session.user.id, organizationId: orgId } }
  })
  if (!membership || membership.role !== "ORG_ADMIN") throw new Error("Only Org Admins can invite")

  const token = crypto.randomBytes(32).toString("hex")
  
  await prisma.organizationInvitation.create({
    data: {
      email,
      organizationId: orgId,
      role,
      token,
      inviterId: session.user.id,
      expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000)
    }
  })

  // Mock Email Send
  console.log(`[MOCK EMAIL] Invite sent to ${email}: ${process.env.NEXT_PUBLIC_APP_URL}/api/org/invite?token=${token}`);
  
  revalidatePath(`/dashboard/organizations/${orgId}`)
}

export async function acceptInvitation(token: string) {
  const session = await auth()
  if (!session?.user?.id || !session.user.email) throw new Error("Unauthorized")

  const invite = await prisma.organizationInvitation.findUnique({
    where: { token }
  })

  if (!invite) throw new Error("Invalid invitation token")
  if (invite.status !== "PENDING") throw new Error("Invitation already processed")
  if (invite.expiresAt < new Date()) throw new Error("Invitation expired")
  if (invite.email !== session.user.email) throw new Error("This invite was sent to a different email")

  await prisma.$transaction([
    prisma.organizationMember.create({
      data: {
        userId: session.user.id,
        organizationId: invite.organizationId,
        role: invite.role
      }
    }),
    prisma.organizationInvitation.update({
      where: { id: invite.id },
      data: { status: "ACCEPTED" }
    })
  ])

  redirect("/dashboard/organizations")
}

2. The Healthcare Navigation Engine (The Core Product)
This module is the brain. It takes the TravelHealthSession, runs it through Triage, checks the Country Profile, pulls Medications/Providers based on the route, and generates a unified Action Plan.

src/modules/navigation/engine.ts

"use server"

import { prisma } from "@/lib/prisma"
import { calculateSeverity } from "@/modules/triage/severity-engine"
import { routeBySeverity, applyCountryOverrides } from "@/modules/triage/router"

export async function runNavigationEngine(sessionId: string) {
  const session = await prisma.travelHealthSession.findUnique({
    where: { id: sessionId },
    include: { user: { include: { memory: true } }, timeline: true }
  })

  if (!session) throw new Error("Session not found")

  // 1. TRIAGE & SEVERITY
  const severity = calculateSeverity({
    symptoms: session.symptoms as string[],
    duration: session.duration || undefined
  })
  
  let route = routeBySeverity(severity)
  route = applyCountryOverrides(route, session.country, session.symptoms as string[])

  // Log Triage Event
  await prisma.journeyTimelineEvent.create({
    data: { sessionId, eventType: "SEVERITY_CALCULATED", data: { severity, route } }
  })

  // 2. COUNTRY CONTEXT
  const countryProfile = await prisma.countryHealthcareProfile.findUnique({
    where: { country: { equals: session.country, mode: "insensitive" } }
  })

  // 3. ACTION PLAN GENERATION (Based on Route)
  let medications: any[] = []
  let providers: any[] = []
  let interpreterContext: any = null

  if (route === "PHARMACY" || route === "CLINIC" || route === "HOSPITAL") {
    // Fetch Meds (Only if not Emergency)
    if (route !== "HOSPITAL") {
      medications = await prisma.countryMedication.findMany({
        where: { country: { equals: session.country, mode: "insensitive" } },
        include: { activeIngredient: true },
        take: 3
      })
    }

    // Fetch Providers based on route
    const providerType = route === "PHARMACY" ? "PHARMACY" : route === "CLINIC" ? "CLINIC" : "HOSPITAL"
    providers = await prisma.provider.findMany({
      where: { 
        country: { equals: session.country, mode: "insensitive" }, 
        city: { equals: session.city, mode: "insensitive" },
        type: providerType
      },
      take: 3
    })

    // Build Interpreter Context
    const memory = session.user.memory
    interpreterContext = {
      severity,
      route,
      symptoms: session.symptoms,
      allergies: memory.filter(m => m.type === "ALLERGY").map(m => m.data),
      medications: memory.filter(m => m.type === "MEDICATION").map(m => m.data),
    }
  }

  // 4. UPDATE SESSION
  const updatedSession = await prisma.travelHealthSession.update({
    where: { id: sessionId },
    data: {
      severityLevel: severity,
      providerOptions: providers,
      localMedicationRecommendations: medications,
      interpreterContext
    }
  })

  // Log Action Plan Created
  await prisma.journeyTimelineEvent.create({
    data: { sessionId, eventType: "ROUTE_ASSIGNED", data: { route, providerCount: providers.length, medCount: medications.length } }
  })

  // 5. RETURN UNIFIED ACTION PLAN
  return {
    severity,
    route,
    confidence: 0.82, // Mock confidence score
    countryProfile,
    actionPlan: {
      medications,
      providers,
      interpreterReady: !!interpreterContext
    },
    session: updatedSession
  }
}

// Specialized Flow: Medication Refill
export async function runRefillFlow(sessionId: string, lostMedicationName: string) {
  // 1. Find active ingredient of the lost medication from Memory or DB
  const userMeds = await prisma.medicalMemory.findMany({ 
    where: { type: "MEDICATION", data: { path: ["name"], string_contains: lostMedicationName } } 
  })
  
  // 2. If we know the ingredient, find local equivalent
  // (Simplified for example)
  const localEquivalent = await prisma.countryMedication.findFirst({
    where: { country: "Japan", activeIngredient: { name: { contains: lostMedicationName } } }
  })

  // 3. Find Pharmacy
  const pharmacy = await prisma.provider.findFirst({
    where: { type: "PHARMACY", country: "Japan" }
  })

  await prisma.journeyTimelineEvent.create({
    data: { sessionId, eventType: "MED_REFILL_FLOW", data: { requested: lostMedicationName, found: localEquivalent?.brandName } }
  })

  return { localEquivalent, pharmacy }
}

3. The New Dashboard: Journey-First UI
Instead of a grid of separate apps, the Dashboard is now a Navigation Command Center. When a user starts a journey, they don't leave the page to go to a "Providers" tab; the Navigation Engine runs and renders the unified Action Plan right there.

src/app/(dashboard)/page.tsx (The new Navigation Dashboard)

import { auth } from "@/lib/auth"
import { prisma } from "@/lib/prisma"
import { NavigationRunner } from "./_components/navigation-runner"

export default async function DashboardHome() {
  const session = await auth()
  
  // Find the most recent active journey
  const activeJourney = await prisma.travelHealthSession.findFirst({
    where: { userId: session!.user.id, outcomeStatus: "ACTIVE" },
    include: { timeline: { orderBy: { createdAt: "asc" } } },
    orderBy: { createdAt: "desc" }
  })

  return (
    <div className="space-y-8">
      <div>
        <h1 className="text-3xl font-bold tracking-tight">Navigation Dashboard</h1>
        <p className="text-muted-foreground mt-1">
          Don't guess what to do next. Let the Navigation OS plot the course.
        </p>
      </div>

      <NavigationRunner activeJourney={activeJourney} />
    </div>
  )
}

src/app/(dashboard)/_components/navigation-runner.tsx

"use client"

import { runNavigationEngine } from "@/modules/navigation/engine"
import { useState } from "react"
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
import { Badge } from "@/components/ui/badge"
import { Button } from "@/components/ui/button"
import { AlertTriangle, Pill, Building2, Languages, MapPin, Phone } from "lucide-react"

export function NavigationRunner({ activeJourney }: { activeJourney: any }) {
  const [plan, setPlan] = useState<any>(null)
  const [loading, setLoading] = useState(false)

  const runEngine = async () => {
    if (!activeJourney) return
    setLoading(true)
    const result = await runNavigationEngine(activeJourney.id)
    setPlan(result)
    setLoading(false)
  }

  if (!activeJourney) {
    return (
      <Card className="border-dashed">
        <CardContent className="py-10 flex flex-col items-center text-center">
          <p className="text-muted-foreground mb-4">No active healthcare journey.</p>
          <Button asChild>
            <a href="/dashboard/journeys/new">Start New Journey</a>
          </Button>
        </CardContent>
      </Card>
    )
  }

  return (
    <div className="space-y-6">
      {/* Journey Context */}
      <Card>
        <CardContent className="p-6 flex justify-between items-center">
          <div>
            <h2 className="font-semibold text-lg flex items-center gap-2">
              <MapPin className="h-5 w-5 text-primary" /> {activeJourney.country}, {activeJourney.city}
            </h2>
            <p className="text-sm text-muted-foreground mt-1">Symptoms: {activeJourney.symptoms.join(", ")}</p>
          </div>
          <Button onClick={runEngine} disabled={loading}>
            {loading ? "Navigating..." : "Run Navigation Engine"}
          </Button>
        </CardContent>
      </Card>

      {/* The Action Plan (Generated by Engine) */}
      {plan && (
        <div className="space-y-6">
          {/* Severity & Route */}
          <div className={`p-4 rounded-lg border-2 ${plan.route === "EMERGENCY" ? "border-destructive bg-destructive/10" : "border-primary bg-primary/5"}`}>
            <div className="flex justify-between items-center mb-2">
              <h3 className="font-bold text-lg flex items-center gap-2">
                {plan.route === "EMERGENCY" && <AlertTriangle className="h-5 w-5 text-destructive" />}
                Recommended Route: {plan.route.replace("_", " ")}
              </h3>
              <Badge variant={plan.severity === "EMERGENCY" ? "destructive" : "default"}>
                Severity: {plan.severity}
              </Badge>
            </div>
            {plan.countryProfile && (
              <p className="text-sm text-muted-foreground flex items-center gap-2">
                <Phone className="h-3 w-3" /> Emergency Number: {plan.countryProfile.emergencyNumber}
              </p>
            )}
          </div>

          {/* The Tools Supporting the Plan */}
          <div className="grid md:grid-cols-2 gap-6">
            {/* Medications Tool */}
            {plan.actionPlan.medications.length > 0 && (
              <Card>
                <CardHeader><CardTitle className="text-base flex items-center gap-2"><Pill className="h-4 w-4" /> Medication Intelligence</CardTitle></CardHeader>
                <CardContent className="space-y-2">
                  {plan.actionPlan.medications.map((med: any) => (
                    <div key={med.id} className="flex justify-between text-sm p-2 border rounded">
                      <span className="font-medium">{med.brandName}</span>
                      <span className="text-muted-foreground">{med.estimatedCost || "Cost N/A"}</span>
                    </div>
                  ))}
                </CardContent>
              </Card>
            )}

            {/* Providers Tool */}
            {plan.actionPlan.providers.length > 0 && (
              <Card>
                <CardHeader><CardTitle className="text-base flex items-center gap-2"><Building2 className="h-4 w-4" /> Provider Discovery</CardTitle></CardHeader>
                <CardContent className="space-y-2">
                  {plan.actionPlan.providers.map((prov: any) => (
                    <div key={prov.id} className="flex justify-between text-sm p-2 border rounded">
                      <span className="font-medium">{prov.name}</span>
                      <span className="text-muted-foreground">{prov.estimatedCost || "Cost N/A"}</span>
                    </div>
                  ))}
                </CardContent>
              </Card>
            )}

            {/* Interpreter Tool */}
            {plan.actionPlan.interpreterReady && (
              <Card className="md:col-span-2 bg-secondary/20">
                <CardContent className="p-6 flex items-center justify-between">
                  <div className="flex items-center gap-3">
                    <Languages className="h-5 w-5 text-primary" />
                    <div>
                      <p className="font-medium">Interpreter Ready</p>
                      <p className="text-sm text-muted-foreground">Context injected: Allergies, Symptoms, Meds.</p>
                    </div>
                  </div>
                  <Button variant="outline">Start Session</Button>
                </CardContent>
              </Card>
            )}
          </div>
        </div>
      )}
    </div>
  )
}
