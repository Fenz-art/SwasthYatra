Based on your explicit architectural correction, here is the strict tech stack to be followed for SwasthYatra. There is no separate backend folder, no Express, no custom JWT middleware, and no Python microservices.

Everything lives inside a single unified Next.js application.

1. Core Framework
Next.js 15/16 (App Router strictly)
React 19 (Server Components by default, Client Components only when interactivity is needed)
TypeScript (Strict mode)
2. Data Fetching & Mutations
Server Actions ("use server") for all mutations (e.g., creating sessions, approving memory candidates, generating passports).
Server Components for direct data fetching (e.g., prisma.findMany() right inside the component).
Route Handlers (app/api/.../route.ts) only for external webhooks (e.g., Deepgram/ElevenLabs callbacks) or public API endpoints (e.g., Provider accessing a Passport link).
3. Database & ORM
PostgreSQL (AWS Aurora PostgreSQL in production)
Prisma ORM (@prisma/client)
Single schema located at prisma/schema.prisma.
4. Authentication & Security
Auth.js v5 (formerly NextAuth)
Google OAuth (Primary provider)
Prisma Adapter (@auth/prisma-adapter)
Cookie-based sessions (JWT strategy for edge compatibility).
Security enforced via the auth() helper inside Server Components and Server Actions.
5. UI & Styling
TailwindCSS (Utility-first CSS)
shadcn/ui (Component architecture using Radix UI primitives)
Lucide React (Icons)
class-variance-authority & tailwind-merge (Component variants)
6. Forms & Validation
React Hook Form (Client-side form management)
Zod (Schema validation on both client and server)
7. AI & Speech Layer (To be integrated in Server Actions)
Vercel AI SDK (For streaming LLM responses and tool calling)
Groq (Llama 4 / DeepSeek R1) → Used for Agent OS Planning, Reasoning, Reflection.
Google Gemini (2.5 Flash/Pro) → Used for Vault document extraction, clinical summaries, and outcome insights.
Deepgram → Speech-to-Text (Interpreter input)
ElevenLabs → Text-to-Speech (Interpreter output)
8. Data Visualization
Recharts → Used for Outcome Intelligence dashboards (Recovery rates, medication efficacy).
9. Infrastructure & Hosting
Vercel (Hosting the Next.js application, edge functions, and serverless functions)
AWS S3 (Storage for Vault Documents, Audio files, Passport PDFs)
AWS Aurora PostgreSQL (Managed database)

❌ express, cors, helmet, morgan

❌ backend/ directory
❌ Python / FastAPI / LangGraph microservices
❌ Fetching internal data via fetch("/api/...") when it can be done directly via Prisma in a Server Component.

src/lib/auth.ts

import NextAuth from "next-auth";
import GoogleProvider from "next-auth/providers/google";
import { PrismaAdapter } from "@auth/prisma-adapter";
import { prisma } from "@/lib/prisma";

export const { handlers, auth, signIn, signOut } = NextAuth({
  adapter: PrismaAdapter(prisma),
  providers: [
    GoogleProvider({
      clientId: process.env.GOOGLE_CLIENT_ID!,
      clientSecret: process.env.GOOGLE_CLIENT_SECRET!,
    }),
  ],
  session: {
    strategy: "jwt", // Using JWT for server actions/edge compatibility
  },
  callbacks: {
    async jwt({ token, user }) {
      if (user) {
        token.id = user.id;
        // Fetch role from DB in production, defaulting to PATIENT
        const dbUser = await prisma.user.findUnique({ where: { email: user.email! }});
        token.role = dbUser?.role || "PATIENT";
        token.onboardingCompleted = dbUser?.onboardingCompleted || false;
      }
      return token;
    },
    async session({ session, token }) {
      if (session.user) {
        session.user.id = token.id as string;
        session.user.role = token.role as string;
        session.user.onboardingCompleted = token.onboardingCompleted as boolean;
      }
      return session;
    },
  },
  pages: {
    signIn: '/auth/signin',
  },
});

// Extend TypeScript types
declare module "next-auth" {
  interface Session {
    user: {
      id: string;
      name?: string | null;
      email?: string | null;
      image?: string | null;
      role: string;
      onboardingCompleted: boolean;
    }
  }
}

  "inline-flex items-center justify-center whitespace-nowrap rounded-md text-sm font-medium ring-offset-background transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring focus-visible:ring-offset-2 disabled:pointer-events-none disabled:opacity-50",

      size: {
        default: "h-10 px-4 py-2",
        sm: "h-9 rounded-md px-3",
        lg: "h-11 rounded-md px-8",
      },
    },
    defaultVariants: { variant: "default", size: "default" },
  }
)

export interface ButtonProps extends React.ButtonHTMLAttributes<HTMLButtonElement>, VariantProps<typeof buttonVariants> {
  asChild?: boolean
}

"use server"

import { auth } from "@/lib/auth"
import { prisma } from "@/lib/prisma"
import { revalidatePath } from "next/cache"

export async function uploadVaultDocument(formData: FormData) {
  const session = await auth()
  if (!session?.user?.id) throw new Error("Unauthorized")

  const type = formData.get("type") as string
  const fileUrl = formData.get("fileUrl") as string
  const fileName = formData.get("fileName") as string

  const doc = await prisma.vaultDocument.create({
    data: { userId: session.user.id, type, fileUrl, fileName, mimeType: "application/pdf" }
  })

  // Mock Pipeline: OCR -> Translate -> Extract -> Create Candidates
  await processDocumentPipeline(doc.id, session.user.id)

  revalidatePath("/dashboard/vault")
  revalidatePath("/dashboard/memory")
}

async function processDocumentPipeline(docId: string, userId: string) {
  const ocrText = "Patient has Hypertension and Diabetes. Prescribed Metformin 500mg."
  await prisma.vaultDocument.update({ where: { id: docId }, data: { status: "OCR_DONE", ocrText } })

  const extractedData = [
    { type: "CONDITION", data: { name: "Hypertension" } },
    { type: "CONDITION", data: { name: "Diabetes" } },
    { type: "MEDICATION", data: { name: "Metformin", dosage: "500mg" } }
  ]

  for (const item of extractedData) {
    await prisma.medicalMemoryCandidate.create({
      data: { userId, documentId: docId, extractedData: item, status: "PENDING" }
    })
  }
  await prisma.vaultDocument.update({ where: { id: docId }, data: { status: "REVIEW_QUEUE" } })
}

"use server"

import { auth } from "@/lib/auth"
import { prisma } from "@/lib/prisma"
import { revalidatePath } from "next/cache"

export async function approveCandidate(candidateId: string) {
  const session = await auth()
  if (!session?.user?.id) throw new Error("Unauthorized")

  const candidate = await prisma.medicalMemoryCandidate.findUnique({ where: { id: candidateId } })
  if (!candidate || candidate.userId !== session.user.id) throw new Error("Invalid candidate")

  await prisma.medicalMemoryCandidate.update({ where: { id: candidateId }, data: { status: "APPROVED", reviewedBy: session.user.id } })

  await prisma.medicalMemory.create({
    data: {
      userId: session.user.id,
      type: candidate.extractedData.type,
      data: candidate.extractedData.data,
      sourceType: "DOCUMENT",
      sourceDocumentId: candidate.documentId,
      verifiedAt: new Date(),
      verifiedBy: session.user.id
    }
  })

  revalidatePath("/dashboard/memory")
}

export async function rejectCandidate(candidateId: string) {
  const session = await auth()
  if (!session?.user?.id) throw new Error("Unauthorized")

  await prisma.medicalMemoryCandidate.updateMany({
    where: { id: candidateId, userId: session.user.id },
    data: { status: "REJECTED", reviewedBy: session.user.id }
  })

  revalidatePath("/dashboard/memory")
}

"use server"

import { auth } from "@/lib/auth"
import { prisma } from "@/lib/prisma"
import { revalidatePath } from "next/cache"
import crypto from "crypto"

export async function generatePassport() {
  const session = await auth()
  if (!session?.user?.id) throw new Error("Unauthorized")

  const memory = await prisma.medicalMemory.findMany({ where: { userId: session.user.id } })
  
  const passportData = {
    conditions: memory.filter(m => m.type === "CONDITION").map(m => m.data),
    allergies: memory.filter(m => m.type === "ALLERGY").map(m => m.data),
    medications: memory.filter(m => m.type === "MEDICATION").map(m => m.data),
  }

  const shareToken = crypto.randomBytes(32).toString("hex")

  const passport = await prisma.healthPassport.create({
    data: {
      userId: session.user.id,
      data: passportData,
      shareToken,
      expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000)
    }
  })

  revalidatePath("/dashboard/passport")
  return passport
}

5. Public Passport Route Handler
To allow providers to access the passport without auth, we use a Route Handler.

src/app/api/passport/public/[token]/route.ts

import { prisma } from "@/lib/prisma"
import { NextResponse } from "next/server"

export async function GET(req: Request, { params }: { params: { token: string } }) {
  const passport = await prisma.healthPassport.findUnique({ where: { shareToken: params.token } })
  
  if (!passport) return NextResponse.json({ error: "Not found" }, { status: 404 })
  if (passport.expiresAt && passport.expiresAt < new Date()) return NextResponse.json({ error: "Link expired" }, { status: 403 })

  await prisma.passportAuditLog.create({
    data: { passportId: passport.id, action: "VIEWED", accessorIp: req.headers.get("x-forwarded-for") || "unknown" }
  })

  return NextResponse.json({ data: passport.data })
}

"use server"

import { auth } from "@/lib/auth"
import { prisma } from "@/lib/prisma"
import { searchMedicationEquivalents } from "@/modules/medication/actions"
import { searchProviders } from "@/modules/provider/actions"

// Mock LLM Planner (In production: Groq / Gemini)
const mockPlanner = async (goal: string) => {
  if (goal.toLowerCase().includes("stomach pain")) {
    return [
      { tool: "searchMedication", args: { ingredient: "Loperamide", country: "Japan" } },
      { tool: "findProvider", args: { country: "Japan", city: "Tokyo", type: "PHARMACY" } }
    ]
  }
  return []
}

// Mock LLM Reflector
const mockReflector = async (results: any[]) => {
  if (results.every(r => r.success)) return "Successfully found medication equivalents and nearby providers."
  return "Failed to find all necessary resources."
}

export async function executeAgentGoal(goal: string, sessionId?: string) {
  const session = await auth()
  if (!session?.user?.id) throw new Error("Unauthorized")

  // 1. Create Task
  const task = await prisma.agentTask.create({
    data: { userId: session.user.id, sessionId, goal, status: "PLANNING" }
  })

  // 2. Plan
  const plan = await mockPlanner(goal)
  await prisma.agentTask.update({ where: { id: task.id }, data: { status: "EXECUTING", plan: plan as any } })

  // 3. Execute (Internal Tool Calling)
  const executionResults: any[] = []
  for (const step of plan) {
    try {
      let result
      if (step.tool === "searchMedication") {
        result = await searchMedicationEquivalents(step.args.ingredient, step.args.country)
      } else if (step.tool === "findProvider") {
        result = await searchProviders(step.args)
      }
      executionResults.push({ tool: step.tool, success: true, data: result })
    } catch (err) {
      executionResults.push({ tool: step.tool, success: false, error: (err as Error).message })
    }
  }

  // 4. Reflect
  const reflection = await mockReflector(executionResults)
  const updatedTask = await prisma.agentTask.update({
    where: { id: task.id },
    data: { status: "COMPLETED", reflection, result: executionResults as any }
  })

  return updatedTask
}

"use server"

import { auth } from "@/lib/auth"
import { prisma } from "@/lib/prisma"

export async function startInterpreterSession(travelHealthSessionId: string, providerLanguage: string) {
  const session = await auth()
  if (!session?.user?.id) throw new Error("Unauthorized")

  const ths = await prisma.travelHealthSession.findUnique({ where: { id: travelHealthSessionId } })
  if (!ths || ths.userId !== session.user.id) throw new Error("Session not found")

  return prisma.interpreterSession.create({
    data: { travelHealthSessionId, patientLanguage: ths.language, providerLanguage }
  })
}

export async function processPatientAudio(interpreterSessionId: string) {
  const session = await auth()
  if (!session?.user?.id) throw new Error("Unauthorized")

  const interp = await prisma.interpreterSession.findUnique({ where: { id: interpreterSessionId }, include: { travelHealthSession: true } })
  if (!interp) throw new Error("Interpreter session not found")

  // 1. Fetch Context
  const memory = await prisma.medicalMemory.findMany({ where: { userId: session.user.id } })
  const context = {
    symptoms: interp.travelHealthSession.symptoms,
    allergies: memory.filter(m => m.type === "ALLERGY").map(m => m.data),
    medications: memory.filter(m => m.type === "MEDICATION").map(m => m.data)
  }

  // 2. Mock STT (Deepgram)
  const originalText = "My stomach hurts and I have a fever."

  // 3. Mock Translation with Context (Gemini)
  const translatedText = `[Translated to ${interp.providerLanguage} with context: Symptoms: ${context.symptoms.join(",")}] -> Translated text here.`

  // 4. Mock TTS (ElevenLabs)
  const audioUrl = `https://mock-elevenlabs.s3.com/${Date.now()}.mp3`

  // 5. Save message
  return prisma.conversationMessage.create({
    data: { interpreterSessionId, speaker: "PATIENT", originalText, translatedText, audioUrl }
  })
}

import { prisma } from "@/lib/prisma"
import { auth } from "@/lib/auth"

export async function getOutcomeInsights() {
  const session = await auth()
  if (!session?.user?.id) return null

  const outcomes = await prisma.outcome.findMany({ where: { userId: session.user.id } })

  // Aggregate Data
  const totalCases = outcomes.length
  const recovered = outcomes.filter(o => o.result === "RECOVERED").length
  const recoveryRate = totalCases > 0 ? ((recovered / totalCases) * 100).toFixed(1) : "0.0"

  const medCounts: Record<string, number> = {}
  outcomes.forEach(o => {
    if (o.medicationUsed) {
      medCounts[o.medicationUsed] = (medCounts[o.medicationUsed] || 0) + 1
    }
  })

  const medicationData = Object.keys(medCounts).map(name => ({ name, usage: medCounts[name] }))
  const recoveryData = [
    { name: "Recovered", value: recovered },
    { name: "No Change", value: outcomes.filter(o => o.result === "NO_CHANGE").length },
    { name: "Worsened", value: outcomes.filter(o => o.result === "WORSENED").length },
  ]

  return { totalCases, recoveryRate, medicationData, recoveryData }
}

src/modules/triage/severity-engine.ts

import { hasRedFlags } from "./red-flags";

export type Severity = "LOW" | "MEDIUM" | "HIGH" | "EMERGENCY";

interface SeverityInput {
  symptoms: string[];
  duration?: string;
}

export function calculateSeverity({ symptoms, duration }: SeverityInput): Severity {
  // 1. Immediate Emergency Check
  if (hasRedFlags(symptoms)) {
    return "EMERGENCY";
  }

  // 2. Duration Parsing (mock logic, could use NLP in production)
  const durationStr = (duration || "").toLowerCase();
  const isLongDuration = durationStr.includes("week") || durationStr.includes("month");
  const isShortDuration = durationStr.includes("hour") || durationStr.includes("day");

  // 3. High Severity Check (Severe pain or long duration)
  const hasSeverePain = symptoms.some(s => s.toLowerCase().includes("severe"));
  if (hasSeverePain || isLongDuration) {
    return "HIGH";
  }

  // 4. Medium Severity Check (Fever, persistent pain)
  const hasFever = symptoms.some(s => s.toLowerCase().includes("fever"));
  const hasPain = symptoms.some(s => s.toLowerCase().includes("pain"));
  if (hasFever || (hasPain && isShortDuration)) {
    return "MEDIUM";
  }

  // 5. Default to Low
  return "LOW";
}

export type Route = "SELF_CARE" | "PHARMACY" | "CLINIC" | "HOSPITAL" | "EMERGENCY";

export function routeBySeverity(severity: Severity): Route {
  switch (severity) {
    case "EMERGENCY":
      return "EMERGENCY";
    case "HIGH":
      return "HOSPITAL";
    case "MEDIUM":
      return "CLINIC";
    case "LOW":
      return "PHARMACY";
    default:
      return "SELF_CARE";
  }
}

export function applyCountryOverrides(route: Route, country: string, symptoms: string[]): Route {
  if (country === "Japan" && symptoms.includes("fever") && route === "CLINIC") {
    return "PHARMACY"; // In Japan, pharmacies often handle fever first
  }
  return route;
}

"use server"

import { prisma } from "@/lib/prisma";
import { calculateSeverity } from "./severity-engine";
import { routeBySeverity, applyCountryOverrides } from "./router";

export async function orchestrateNavigation(sessionId: string) {
  // 1. Fetch the TravelHealthSession
  const session = await prisma.travelHealthSession.findUnique({
    where: { id: sessionId },
    include: { user: { include: { memory: true } } }
  });

  if (!session) throw new Error("Session not found");

  // 2. Calculate Severity & Route
  const severity = calculateSeverity({
    symptoms: session.symptoms as string[],
    duration: session.duration || undefined
  });
  
  let route = routeBySeverity(severity);
  route = applyCountryOverrides(route, session.country, session.symptoms as string[]);

  // 3. Execute Route-Specific Actions
  let medicationRecommendations: any[] = [];
  let providerRecommendations: any[] = [];

  if (route === "PHARMACY" || route === "CLINIC") {
    // Mock: Fetch medication equivalents based on symptoms
    // In production, map symptoms -> active ingredients -> country equivalents
    medicationRecommendations = await prisma.countryMedication.findMany({
      where: { country: session.country },
      include: { activeIngredient: true },
      take: 3
    });

    // Fetch Providers
    providerRecommendations = await prisma.provider.findMany({
      where: { 
        country: session.country, 
        city: session.city,
        type: route === "PHARMACY" ? "PHARMACY" : "CLINIC"
      },
      take: 5
    });
  }

  if (route === "HOSPITAL" || route === "EMERGENCY") {
    providerRecommendations = await prisma.provider.findMany({
      where: { 
        country: session.country, 
        city: session.city,
        type: "HOSPITAL"
      },
      take: 5
    });
  }

  // 4. Update Session with Navigation Context
  const updatedSession = await prisma.travelHealthSession.update({
    where: { id: sessionId },
    data: {
      severityLevel: severity,
      providerOptions: providerRecommendations,
      localMedicationRecommendations: medicationRecommendations,
      interpreterContext: {
        severity,
        route,
        patientHistory: session.user.memory
      }
    }
  });

  return {
    severity,
    route,
    medications: medicationRecommendations,
    providers: providerRecommendations,
    session: updatedSession
  };
}

// Define the signature for an Agent Tool
type AgentTool = (args: any, context: { userId: string, sessionId?: string }) => Promise<any>;

// The Registry
export const toolRegistry: Record<string, AgentTool> = {
  
  queryPassport: async (_, ctx) => {
    return prisma.healthPassport.findFirst({ 
      where: { userId: ctx.userId },
      orderBy: { createdAt: "desc" }
    });
  },

  queryMemory: async (_, ctx) => {
    return prisma.medicalMemory.findMany({ where: { userId: ctx.userId } });
  },

  searchMedication: async (args, _) => {
    return prisma.countryMedication.findMany({
      where: {
        activeIngredient: { name: { equals: args.ingredient, mode: "insensitive" } },
        country: { equals: args.country, mode: "insensitive" }
      },
      include: { activeIngredient: true }
    });
  },

  findProvider: async (args, _) => {
    return prisma.provider.findMany({
      where: {
        country: { equals: args.country, mode: "insensitive" },
        city: { equals: args.city, mode: "insensitive" },
        type: args.type
      }
    });
  },

  // The ultimate tool: The Navigation Router
  navigateSymptoms: async (args, ctx) => {
    if (!ctx.sessionId) throw new Error("Session required for navigation");
    return orchestrateNavigation(ctx.sessionId);
  }
};

export async function executeTool(toolName: string, args: any, ctx: { userId: string, sessionId?: string }) {
  const tool = toolRegistry[toolName];
  if (!tool) throw new Error(`Tool ${toolName} not found in registry`);
  return tool(args, ctx);
}

export async function buildInterpreterContext(sessionId: string) {

    - Known Allergies: ${memory.filter(m => m.type === "ALLERGY").map(m => (m.data as any).name).join(", ") || "None"}

    Ensure medical terminology is preserved. Highlight any potential conflicts between spoken symptoms and existing allergies/medications.
  `;

    textContext: contextString,

"use server"

import { prisma } from "@/lib/prisma"
import { auth } from "@/lib/auth"

export async function getGlobalInsights(symptoms: string[], country: string) {
  const session = await auth()
  if (!session?.user?.id) return null

  // Find historical outcomes matching these symptoms in this country
  const outcomes = await prisma.outcome.findMany({
    where: {
      country: { equals: country, mode: "insensitive" },
      symptoms: { hasSome: symptoms }
    }
  })

  if (outcomes.length === 0) {
    return {
      totalCases: 0,
      recoveryRate: "0.0",
      avgRecoveryHours: 0,
      medicationData: [],
      recoveryData: [],
      providerData: []
    }
  }

  // 1. Basic Metrics
  const totalCases = outcomes.length
  const recoveredCases = outcomes.filter(o => o.result === "RECOVERED")
  const recoveryRate = ((recoveredCases.length / totalCases) * 100).toFixed(1)

  // 2. Recovery Time
  const validRecoveryTimes = recoveredCases.filter(o => o.recoveryTimeHours)
  const avgRecoveryHours = validRecoveryTimes.length > 0 
    ? Math.round(validRecoveryTimes.reduce((sum, o) => sum + (o.recoveryTimeHours || 0), 0) / validRecoveryTimes.length)
    : 0

  // 3. Medication Efficacy
  const medCounts: Record<string, { success: number; total: number }> = {}
  outcomes.forEach(o => {
    if (o.medicationUsed) {
      if (!medCounts[o.medicationUsed]) medCounts[o.medicationUsed] = { success: 0, total: 0 }
      medCounts[o.medicationUsed].total++
      if (o.result === "RECOVERED") medCounts[o.medicationUsed].success++
    }
  })
  
  const medicationData = Object.keys(medCounts).map(name => ({
    name,
    usage: medCounts[name].total,
    successRate: medCounts[name].total > 0 
      ? `${((medCounts[name].success / medCounts[name].total) * 100).toFixed(0)}%` 
      : "N/A"
  }))

  // 4. Recovery Distribution (for Pie Chart)
  const recoveryData = [
    { name: "Recovered", value: recoveredCases.length },
    { name: "No Change", value: outcomes.filter(o => o.result === "NO_CHANGE").length },
    { name: "Worsened", value: outcomes.filter(o => o.result === "WORSENED").length },
  ]

  // 5. Provider Success Rate
  const providerCounts: Record<string, { success: number; total: number }> = {}
  outcomes.forEach(o => {
    if (o.providerType) {
      if (!providerCounts[o.providerType]) providerCounts[o.providerType] = { success: 0, total: 0 }
      providerCounts[o.providerType].total++
      if (o.result === "RECOVERED") providerCounts[o.providerType].success++
    }
  })

  const providerData = Object.keys(providerCounts).map(type => ({
    name: type,
    successRate: providerCounts[type].total > 0 
      ? ((providerCounts[type].success / providerCounts[type].total) * 100).toFixed(1) 
      : "0.0"
  }))

  return {
    totalCases,
    recoveryRate,
    avgRecoveryHours,
    medicationData,
    recoveryData,
    providerData
  }
}

export async function generatePresignedUploadUrl(fileName: string, fileType: string) {
  const key = `vault/${Date.now()}-${fileName}`;
  const command = new PutObjectCommand({
    Bucket: process.env.AWS_S3_BUCKET_NAME!,
    Key: key,
    ContentType: fileType,
  });

  const url = await getSignedUrl(s3, command, { expiresIn: 60 }); // 1 minute expiry
  return { url, key };
}

import PDFDocument from "pdfkit";
import QRCode from "qrcode";
import { prisma } from "@/lib/prisma";

export async function generatePassportPDF(passportId: string): Promise<Buffer> {
  const passport = await prisma.healthPassport.findUnique({
    where: { id: passportId },
    include: { user: true }
  });

  if (!passport) throw new Error("Passport not found");

  const doc = new PDFDocument({ margin: 50 });
  const buffers: Buffer[] = [];
  doc.on("data", buffers.push.bind(buffers));

  return new Promise<Buffer>(async (resolve, reject) => {
    doc.on("end", () => {
      const pdfData = Buffer.concat(buffers);
      resolve(pdfData);
    });

    doc.on("error", reject);

    // Header
    doc.fontSize(20).fillColor("#2563eb").text("SwasthYatra Health Passport", { align: "center" });
    doc.moveDown();
    doc.fontSize(12).fillColor("#000").text(`Patient: ${passport.user.name}`, { align: "center" });
    doc.moveDown(2);

    // QR Code Generation
    const shareUrl = `${process.env.NEXT_PUBLIC_APP_URL}/api/passport/public/${passport.shareToken}`;
    const qrBuffer = await QRCode.toBuffer(shareUrl, { width: 150, margin: 1 });
    
    doc.image(qrBuffer, { fit: [150, 150], align: "center" });
    doc.moveDown(2);

    const data = passport.data as any;

    // Conditions
    doc.fontSize(14).fillColor("#dc2626").text("Conditions", { underline: true });
    doc.fontSize(11).fillColor("#000");
    data.conditions.forEach((c: any) => doc.text(`- ${c.name}`));
    doc.moveDown();

    // Allergies
    doc.fontSize(14).fillColor("#dc2626").text("Allergies", { underline: true });
    doc.fontSize(11).fillColor("#000");
    if (data.allergies.length === 0) doc.text("None reported");
    data.allergies.forEach((a: any) => doc.text(`- ${a.name}`));
    doc.moveDown();

    // Medications
    doc.fontSize(14).fillColor("#2563eb").text("Medications", { underline: true });
    doc.fontSize(11).fillColor("#000");
    if (data.medications.length === 0) doc.text("None reported");
    data.medications.forEach((m: any) => doc.text(`- ${m.name} ${m.dosage || ""}`));
    doc.moveDown();

    // Footer
    doc.fontSize(9).fillColor("#666").text(`Generated: ${new Date(passport.createdAt).toLocaleString()}`, { align: "center" });
    doc.text(`Expires: ${passport.expiresAt ? new Date(passport.expiresAt).toLocaleString() : 'N/A'}`, { align: "center" });

    doc.end();
  });
}

  const emergencyRelation = formData.get("emergencyRelation") as string
  const insuranceProvider = formData.get("insuranceProvider") as string
  const insurancePolicy = formData.get("insurancePolicy") as string
  
  // Medical basics
  const conditions = (formData.get("conditions") as string).split(",").map(s => s.trim()).filter(Boolean)
  const allergies = (formData.get("allergies") as string).split(",").map(s => s.trim()).filter(Boolean)
  const medications = (formData.get("medications") as string).split(",").map(s => s.trim()).filter(Boolean)

  // 1. Update User Profile
  await prisma.user.update({
    where: { id: session.user.id },
    data: {
      onboardingCompleted: true,
      bloodType,
      emergencyContact: { name: emergencyName, phone: emergencyPhone, relation: emergencyRelation },
      insuranceInfo: { provider: insuranceProvider, policyNumber: insurancePolicy }
    }
  })

  // 2. Auto-populate Medical Memory (High Confidence)
  const memoryOps = []
  
  conditions.forEach(name => {
    memoryOps.push(prisma.medicalMemory.create({
      data: { userId: session.user.id, type: "CONDITION", data: { name }, sourceType: "ONBOARDING", confidence: 1.0, verifiedAt: new Date() }
    }))
  })

  allergies.forEach(name => {
    memoryOps.push(prisma.medicalMemory.create({
      data: { userId: session.user.id, type: "ALLERGY", data: { name }, sourceType: "ONBOARDING", confidence: 1.0, verifiedAt: new Date() }
    }))
  })

  medications.forEach(name => {
    memoryOps.push(prisma.medicalMemory.create({
      data: { userId: session.user.id, type: "MEDICATION", data: { name }, sourceType: "ONBOARDING", confidence: 1.0, verifiedAt: new Date() }
    }))
  })

  await Promise.all(memoryOps)

  revalidatePath("/dashboard")
  redirect("/dashboard")
}

3. Organization Workflows (RBAC & Invitations)
This module handles creating organizations, inviting members via secure tokens, and accepting invitations.

src/modules/organization/actions.ts

"use server"

import { auth } from "@/lib/auth"
import { prisma } from "@/lib/prisma"
import { revalidatePath } from "next/cache"
import crypto from "crypto"
import { redirect } from "next/navigation"

export async function createOrganization(name: string, type: string) {
  const session = await auth()
  if (!session?.user?.id) throw new Error("Unauthorized")

  const org = await prisma.organization.create({
    data: {
      name,
      type,
      members: {
        create: {
          userId: session.user.id,
          role: "ORG_ADMIN"
        }
      }
    }
  })

  revalidatePath("/dashboard/organizations")
  return org
}

export async function inviteMember(email: string, role: string, orgId: string) {
  const session = await auth()
  if (!session?.user?.id) throw new Error("Unauthorized")

  // Verify inviter is Org Admin
  const membership = await prisma.organizationMember.findUnique({
    where: { userId_organizationId: { userId: session.user.id, organizationId: orgId } }
  })
  if (!membership || membership.role !== "ORG_ADMIN") throw new Error("Only Org Admins can invite")

  const token = crypto.randomBytes(32).toString("hex")
  
  await prisma.organizationInvitation.create({
    data: {
      email,
      organizationId: orgId,
      role,
      token,
      inviterId: session.user.id,
      expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000)
    }
  })

  // Mock Email Send
  console.log(`[MOCK EMAIL] Invite sent to ${email}: ${process.env.NEXT_PUBLIC_APP_URL}/api/org/invite?token=${token}`);
  
  revalidatePath(`/dashboard/organizations/${orgId}`)
}

export async function acceptInvitation(token: string) {
  const session = await auth()
  if (!session?.user?.id || !session.user.email) throw new Error("Unauthorized")

  const invite = await prisma.organizationInvitation.findUnique({
    where: { token }
  })

  if (!invite) throw new Error("Invalid invitation token")
  if (invite.status !== "PENDING") throw new Error("Invitation already processed")
  if (invite.expiresAt < new Date()) throw new Error("Invitation expired")
  if (invite.email !== session.user.email) throw new Error("This invite was sent to a different email")

  await prisma.$transaction([
    prisma.organizationMember.create({
      data: {
        userId: session.user.id,
        organizationId: invite.organizationId,
        role: invite.role
      }
    }),
    prisma.organizationInvitation.update({
      where: { id: invite.id },
      data: { status: "ACCEPTED" }
    })
  ])

  redirect("/dashboard/organizations")
}
